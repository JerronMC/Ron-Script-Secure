local function draggable(handle, target)
    local dragging, dragStart, startPos, dragInput
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local d = input.Position - dragStart
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function ttsOpenAI(text)
    local key = Data.keys.OpenAI
    if not key or key == "" then return false, "\079\112\101\110\065\073\032\107\101\121\032\114\101\113\117\105\114\101\100\046" end
    if not Request or not hasFileAPI() or not GetCustomAsset then return false, "\065\080\073\115\032\109\105\115\115\105\110\103\046" end
    local body = { model = "\103\112\116\045\052\111\045\109\105\110\105\045\116\116\115", voice = Data.ttsVoice or "\097\108\108\111\121",
        input = text, response_format = "\109\112\051" }
    local res, err = doRequest({
        Url = "\104\116\116\112\115\058\047\047\097\112\105\046\111\112\101\110\097\105\046\099\111\109\047\118\049\047\097\117\100\105\111\047\115\112\101\101\099\104", Method = "\080\079\083\084",
        Headers = { ["\067\111\110\116\101\110\116\045\084\121\112\101"] = "\097\112\112\108\105\099\097\116\105\111\110\047\106\115\111\110", ["\065\117\116\104\111\114\105\122\097\116\105\111\110"] = "\066\101\097\114\101\114\032" .. key },
        Body = HttpService:JSONEncode(body) })
    if not res then return false, err end
    local file = SAVE_FOLDER .. "\047\118\111\105\099\101\046\109\112\051"
    local ok = pcall(function()
        if hasFolderAPI() and not isfolder(SAVE_FOLDER) then makefolder(SAVE_FOLDER) end
        writefile(file, res.Body) end)
    if not ok then return false, "\083\097\118\101\032\102\097\105\108\101\100\046" end
    local ok2, asset = pcall(function() return GetCustomAsset(file) end)
    if not ok2 or not asset then return false, "\076\111\097\100\032\102\097\105\108\101\100\046" end
    local old = GuiParent:FindFirstChild("\067\104\097\116\071\080\084\095\084\084\083\095\082\111\110\110\105\101")
    if old then pcall(function() old:Stop(); old:Destroy() end) end
    local s = create("\083\111\117\110\100", { Parent = GuiParent, Name = "\067\104\097\116\071\080\084\095\084\084\083\095\082\111\110\110\105\101",
        SoundId = asset, Volume = 0x1 })
    local ok3 = pcall(function() s:Play() end)
    if not ok3 then s:Destroy(); return false, "\080\108\097\121\098\097\099\107\032\114\101\102\117\115\101\100\046" end
    return true
end

--// SEND
local Sending = false
local function sendMessage()
    if Sending then return end
    local prompt = Prompt.Text
    if not prompt or prompt:gsub("\037\115\043","") == "" then return end
    local activeKey = Data.keys[Data.activeProvider]
    if not activeKey or activeKey == "" then
        Settings.Visible = true; notify("\065\100\100\032\097\032" .. Data.activeProvider .. "\032\107\101\121", "\119\097\114\110\105\110\103"); return
    end
    if Data.activeProvider == "\067\117\115\116\111\109\032\047\032\079\112\101\110\065\073\045\067\111\109\112\097\116\105\098\108\101"
        and (not Data.customBaseURL or Data.customBaseURL == "") then
        Settings.Visible = true; notify("\065\100\100\032\099\117\115\116\111\109\032\098\097\115\101\032\085\082\076", "\119\097\114\110\105\110\103"); return
    end
    Sending = true; Prompt.Text = ""
    -- pulse send
    if Data.animationsEnabled then
        tween(SendBtn, { Size = UDim2.new(0x0, 0x22, 0x0, 0x22) }, 0.08)
        task.delay(0.1, function() tween(SendBtn, { Size = UDim2.new(0x0, 0x26, 0x0, 0x26) }, 0.2, Enum.EasingStyle.Back) end)
    end
    CurrentChat = getCurrentChat()
    if CurrentChat.title == "\078\101\119\032\067\104\097\116" then
        CurrentChat.title = safeText(prompt:gsub("\037\115\043","\032"), 0x22)
    end
    CurrentChat.provider = Data.activeProvider
    CurrentChat.model = selectedModel()
    CurrentChat.updatedAt = os.time()
    table.insert(CurrentChat.messages, { role = "\117\115\101\114", content = prompt, time = os.time() })
    saveData(); rebuildRecent()
    renderChat()
    scrollToBottom(true)
    local stop = thinkingIndicator()
    task.spawn(function()
        local answer, err = askAI(prompt)
        stop()
        if answer then
            table.insert(CurrentChat.messages, { role = "\097\115\115\105\115\116\097\110\116", content = answer, time = os.time() })
            CurrentChat.updatedAt = os.time()
            saveData(); renderChat(); rebuildRecent()
            scrollToBottom(true)
            if Data.voiceRead and _G.ChatGPTRonnieReadVoice then
                _G.ChatGPTRonnieReadVoice(answer)
            end
        else
            table.insert(CurrentChat.messages, { role = "\097\115\115\105\115\116\097\110\116",
                content = "\091\069\114\114\111\114\093\032" .. tostring(err or "\082\101\113\117\101\115\116\032\102\097\105\108\101\100\046"), time = os.time() })
            saveData(); renderChat()
            scrollToBottom(true)
            notify(tostring(err or "\082\101\113\117\101\115\116\032\102\097\105\108\101\100"), "\101\114\114\111\114")
        end
        Sending = false
    end)
end
